# SovereignAI API Endpoints - cURL Commands for Postman Import
# Base URL: http://localhost:8000

# --- General ---

# Health Check
curl -X GET "http://localhost:8000/"

# Detailed Health Check
curl -X GET "http://localhost:8000/health"

# --- Chat ---

# Chat Completions
curl -X POST "http://localhost:8000/v1/chat/completions" \
     -H "Content-Type: application/json" \
     -d '{
           "messages": [
             {"role": "system", "content": "You are a helpful assistant."},
             {"role": "user", "content": "Hello!"}
           ],
           "model": "llama3:8b",
           "max_tokens": 512,
           "temperature": 0.7,
           "stream": false,
           "use_rag": false
         }'

# Universal Execution
curl -X POST "http://localhost:8000/v1/chat/execute" \
     -H "Content-Type: application/json" \
     -d '{
           "task": "summarize",
           "input": "This is a long text to summarize.",
           "generation_kwargs": {
             "max_tokens": 100
           }
         }'

# Switch Execution Mode
curl -X POST "http://localhost:8000/v1/chat/mode/switch?mode=layerstream"

# --- Models ---

# List Models
curl -X GET "http://localhost:8000/v1/models/"

# Pull Model
curl -X POST "http://localhost:8000/v1/models/pull" \
     -H "Content-Type: application/json" \
     -d '{
           "model": "phi-2",
           "quant": "q4_k_m"
         }'

# Pull Status
curl -X GET "http://localhost:8000/v1/models/pull/status/phi-2"

# Load Model
curl -X POST "http://localhost:8000/v1/models/load" \
     -H "Content-Type: application/json" \
     -d '{
           "model": "phi-2",
           "mode": "fullram"
         }'

# Unload Model
curl -X POST "http://localhost:8000/v1/models/unload"

# Refresh Models
curl -X POST "http://localhost:8000/v1/models/refresh"

# Current Loaded Model
curl -X GET "http://localhost:8000/v1/models/current"

# Get Model Details
curl -X GET "http://localhost:8000/v1/models/phi-2"

# Delete Model
curl -X DELETE "http://localhost:8000/v1/models/phi-2"
